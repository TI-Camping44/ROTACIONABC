Apto para disparador. */ function stkAutomatico() { const inicio = new Date(); try { const datos = stkActualizarDatos(); stkPublicar(datos); PropertiesService.getScriptProperties() .setProperty('STK_ULTIMA_CORRIDA', `${stkAhora_()} · OK`); Logger.log(`Actualización completa en ${Math.round((new Date() - inicio)/1000)} s.`); } catch (error) { const mensaje = error && error.message ? error.message : String(error); PropertiesService.getScriptProperties() .setProperty('STK_ULTIMA_CORRIDA', `${stkAhora_()} · ERROR: ${mensaje.substring(0, 200)}`); Logger.log(`FALLÓ: ${mensaje}`); throw error; /* que quede registrado en el panel de ejecuciones */ } } function stkProgramar() { const id = stkSS_().getId(); PropertiesService.getScriptProperties().setProperty('SPREADSHEET_ID', id); stkQuitarDisparadores_(); ScriptApp.newTrigger('stkAutomatico') .timeBased() .atHour(STK_HORA_ACTUALIZACION) .everyDays(1) .inTimezone(STK.ZONA_HORARIA) .create(); stkAvisar_('Actualización programada', [ `Todos los días alrededor de las ${String(STK_HORA_ACTUALIZACION).padStart(2, '0')}:00.`, '', 'Extrae de Odoo y publica el reporte, en ese orden.', 'Si falla, no publica nada: el reporte queda con los últimos datos buenos.', '', 'Google puede correrlo hasta una hora antes o después. 'Programación cancelada.' : 'No había ninguna programación activa.'); } function stkQuitarDisparadores_() { let cantidad = 0; ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'stkAutomatico') { ScriptApp.deleteTrigger(t); cantidad += 1; } }); return cantidad; } function stkEstadoProgramacion() { const activos = ScriptApp.getProjectTriggers() .filter(t => t.getHandlerFunction() === 'stkAutomatico').length; const ultima = PropertiesService.getScriptProperties().getProperty('STK_ULTIMA_CORRIDA'); stkAvisar_('Estado de la actualización', [ `Programación: ${activos ? Ejecutá "3. Actualizar datos".`); return h.getDataRange().getValues().slice(1); } function stkEscribir_(nombre, titulos, filas, columnasNumericas) { const h = stkHoja_(nombre); if (h.getFilter()) h.getFilter().remove(); h.setConditionalFormatRules([]); h.clear(); /* Formato TEXTO en todo lo que no sea número, ANTES de escribir.
